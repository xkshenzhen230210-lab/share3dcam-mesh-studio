cmake_minimum_required(VERSION 3.20)
project(mesh_writer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(mesh_writer writer.cpp)
target_include_directories(mesh_writer PRIVATE "${OSG_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../vendor")
target_link_directories(mesh_writer PRIVATE "${OSG_ROOT}/lib")
target_link_libraries(mesh_writer PRIVATE osg osgDB OpenThreads)
target_compile_options(mesh_writer PRIVATE /utf-8)
foreach(name reader viewer render)
  add_executable(mesh_${name} ${name}.cpp)
  target_include_directories(mesh_${name} PRIVATE "${OSG_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../vendor")
  target_link_directories(mesh_${name} PRIVATE "${OSG_ROOT}/lib")
  target_link_libraries(mesh_${name} PRIVATE osg osgDB osgUtil osgGA osgViewer OpenThreads)
  target_compile_options(mesh_${name} PRIVATE /utf-8)
  target_link_options(mesh_${name} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/utf8.manifest")
endforeach()
target_link_libraries(mesh_render PRIVATE opengl32)
